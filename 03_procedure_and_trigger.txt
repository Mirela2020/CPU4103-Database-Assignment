USE CrownfieldEyeHospital;

-- ============================================================
-- TASK 1: STORED PROCEDURE
-- Retrieve appointments for a specified patient ID.
-- ============================================================

DROP PROCEDURE IF EXISTS GetPatientAppointment;

DELIMITER $$
CREATE PROCEDURE GetPatientAppointment(IN p_patient_id INT)
BEGIN
    SELECT appointment_id, patient_id
    FROM Appointment
    WHERE patient_id = p_patient_id;
END$$
DELIMITER ;

-- Test using an existing patient ID (1 was used in the report):
CALL GetPatientAppointment(1);

-- ============================================================
-- TASK 2: TRIGGER
-- Automatically set a new appointment's status to 'Scheduled'.
-- Requires an Appointment.status column (as in the report's test).
-- ============================================================

DROP TRIGGER IF EXISTS SetAppointmentStatus;

DELIMITER $$
CREATE TRIGGER SetAppointmentStatus
BEFORE INSERT ON Appointment
FOR EACH ROW
BEGIN
    SET NEW.status = 'Scheduled';
END$$
DELIMITER ;

-- specifying status, then checking that status was 'Scheduled'.
